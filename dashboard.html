const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwCc3ZFIb2Tkp4ZA971WpOHWBgYz3NiWhK4bLYxk99_a4nK9XX9mYSuVRSnPlUs-tjDeg/exec"; let isFetching = false; // Duplicate requests rokne ke liye async function fetchLiveData() { if (isFetching) return; isFetching = true; try { // Cache Busting parameter (?t=timestamp) taaki Browser purana data na dikhaye const response = await fetch(`${SCRIPT_URL}?t=${new Date().getTime()}`, { method: 'GET', redirect: 'follow' }); if (!response.ok) throw new Error("Network issue"); const data = await response.json(); if (Array.isArray(data)) { renderTable(data); document.getElementById('status-indicator').innerText = "🟢 Live Connected"; document.getElementById('status-indicator').style.color = "green"; } } catch (error) { console.warn("Live update retry error:", error); document.getElementById('status-indicator').innerText = "🟡 Reconnecting..."; document.getElementById('status-indicator').style.color = "orange"; } finally { isFetching = false; } } function renderTable(data) { const tbody = document.getElementById('tableBody'); if (data.length === 0) { tbody.innerHTML = "Koi Data Nahi Hai"; return; } let rowsHtml = ""; data.forEach(item => { rowsHtml += ` ${item.name || ''} ${item.email || ''} ${item.message || ''} ${item.date || item.timestamp || ''} `; }); tbody.innerHTML = rowsHtml; document.getElementById('loading').style.display = "none"; document.getElementById('dataTable').style.display = "table"; } // 1. Page khulte hi pehli baar data layega fetchLiveData(); // 2. Har 10 Seconds (10000ms) me Bina Rukavat Auto Live Update Hoga setInterval(fetchLiveData, 10000);
