const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwCc3ZFIb2Tkp4ZA971WpOHWBgYz3NiWhK4bLYxk99_a4nK9XX9mYSuVRSnPlUs-tjDeg/exec"; let isFetching = false; let globalSheetData = []; // Pure dashboard ke filter ke liye global memory // JSONP Fetch Method (CORS Proof) function fetchLiveData() { if (isFetching) return; isFetching = true; // Purani script file ko remove karna taaki DOM clean rahe const oldScript = document.getElementById('jsonp-script'); if (oldScript) oldScript.remove(); // Dynamically script insert karke data fetch karna const script = document.createElement('script'); script.id = 'jsonp-script'; script.src = `${SCRIPT_URL}?callback=handleServerResponse&t=${new Date().getTime()}`; script.onerror = function() { isFetching = false; updateStatus("🟡 Reconnecting...", "orange"); }; document.body.appendChild(script); } // Google Sheet se Data milne par ye automatically call hoga function handleServerResponse(data) { isFetching = false; if (Array.isArray(data) && data.length > 0) { globalSheetData = data; // Check agar aapke dashboard par applyFilters() hai toh use chalao, nahi toh direct render if (typeof applyFilters === "function") { applyFilters(); } else { renderTable(globalSheetData); } updateStatus("🟢 Live Connected", "#05cd99"); } else { renderTable([]); updateStatus("🟢 Connected (No Data)", "#05cd99"); } } // Status message updater with safety check function updateStatus(text, color) { const statusElem = document.getElementById('status-indicator'); if (statusElem) { statusElem.innerText = text; statusElem.style.color = color; } } function renderTable(data) { const tbody = document.getElementById('tableBody'); if (!tbody) return; if (!data || data.length === 0) { tbody.innerHTML = "Koi Data Nahi Hai"; return; } let rowsHtml = ""; data.forEach((item, index) => { rowsHtml += ` ${index + 1} ${item.name || item['Vehicle Reg No'] || 'N/A'} ${item.email || item.Owner || 'N/A'} ${item.message || item.Remarks || 'N/A'} ${item.date || item.timestamp || 'N/A'} `; }); tbody.innerHTML = rowsHtml; // Loader and Table display safety checks const loader = document.getElementById('loading') || document.getElementById('loader'); const table = document.getElementById('dataTable') || document.getElementById('tableWrapper'); if (loader) loader.style.display = "none"; if (table) table.style.display = "table"; } // Initial Fetch & Auto 10-Second Refresh Loop fetchLiveData(); setInterval(fetchLiveData, 10000);
